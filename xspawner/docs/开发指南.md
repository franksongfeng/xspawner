  def caller_module(idx=1):
      # stack[0]    caller_module() 自己
      # stack[1]    init_database()（它调用了 caller_module()）
      # stack[2]    调用 init_database() 的那个函数（最终用户代码）
      caller_frame = inspect.stack()[idx]
      return inspect.getmodule(caller_frame[0])
  
  
  def make_connection_str(setting: dict) -> str:
      if "category" not in setting:
          raise ValueError(f"Miss category parameter in setting {setting}")
      category = setting["category"]
  
      conn_str = ""
      if category == "sqlite":
          conn_str = '{}://{}'.format(
              category,
              setting["file"]
              )
      elif category in ("mysql", "postgres"):
          required = ["usr", "psw", "host", "port", "name"]
          missing = [k for k in required if k not in setting]
          if missing:
              raise ValueError(f"Miss parameter: {missing}")
          else:
              conn_str = '{}://{}:{}@{}:{}/{}'.format(
                  category,
                  quote_plus(setting["usr"]),
                  quote_plus(setting["psw"]),
                  setting["host"],
                  setting["port"],
                  quote_plus(setting["name"])
                  )
      else:
          raise ValueError(f"Invalid category: {category}")
      return conn_str
  
  
  def parse_connection_str(conn_str: str) -> dict:
      """
      将 make_connection_str 生成的连接字符串反解析为 settings dict。
  
      输入示例:
        "sqlite://kv.db"
        "sqlite:///opt/data/kv.db"
        "mysql://user:pass@127.0.0.1:3306/mydb"
        "postgres://user%40corp:p%2Fw@db.local:5432/prod"
  
      返回:
        {"category": "sqlite", "file": "kv.db"}
        {"category": "mysql", "usr": "user", "psw": "pass",
         "host": "127.0.0.1", "port": 3306, "name": "mydb"}
      """
      if not isinstance(conn_str, str) or "://" not in conn_str:
          raise ValueError(f"Invalid conn_str: {conn_str!r}")
  
      category, rest = conn_str.split("://", 1)
      category = category.lower()
  
      # ── sqlite: 剩余部分就是文件路径 ─────────────────────
      if category == "sqlite":
          if not rest:
              raise ValueError(f"Empty sqlite file path: {conn_str!r}")
          return {"category": "sqlite", "file": rest}
  
      # ── mysql / postgres: usr:psw@host:port/name ─────────
      if category in ("mysql", "postgres"):
          # usr/psw/name 已被 quote_plus 编码, 因此不含未编码的 : @ /
          m = re.fullmatch(
              r"(?P<usr>[^:@/]+):(?P<psw>[^@]*)@"
              r"(?P<host>[^:/]+):(?P<port>\d+)"
              r"/(?P<name>.+)",
              rest,
          )
          if not m:
              raise ValueError(f"Invalid {category} conn_str: {conn_str!r}")
          g = m.groupdict()
          return {
              "category": category,
              "usr":  unquote_plus(g["usr"]),
              "psw":  unquote_plus(g["psw"]),
              "host": g["host"],           # host 未编码, 原样返回
              "port": int(g["port"]),      # 转 int 便于直接用
              "name": unquote_plus(g["name"]),
          }
      raise ValueError(f"Invalid category: {category}")
  
  
  async def open_database(conn: str, mmod: Optional[str] = None):
      '''
      初始化连接并建表
      (去 orm 模块中查找模型类)
      '''
      await Tortoise.init(
          db_url=conn,
          modules={
              'models': [__name__, mmod] if mmod else [__name__]
          }
      )
  
      await Tortoise.generate_schemas(safe=True)
  
      # 如果是 SQLite，启用 WAL 模式并设置 busy_timeout
      if get_database_category() == "sqlite":
          await get_database_connection().execute_query("PRAGMA journal_mode=WAL;")
          await get_database_connection().execute_query("PRAGMA busy_timeout=5000;")  # 可选：等待锁超时 5 秒
  
  
  def get_database_connection():
      return Tortoise.get_connection("default")
  
  def get_database_category() -> str:
      conn_obj = get_database_connection()
      return conn_obj.capabilities.dialect    # "sqlite" / "mysql" / "postgres"
  
  
  async def close_database():
      '''
      关闭所有数据库连接
      '''
      await Tortoise.close_connections()
  
  
  def _quote_mysql_ident(name: str) -> str:
      """MySQL 标识符转义：反引号包裹，内部反引号翻倍"""
      return "`" + name.replace("`", "``") + "`"
  
  
  def _quote_pg_ident(name: str) -> str:
      """PostgreSQL 标识符转义：双引号包裹，内部双引号翻倍"""
      return '"' + name.replace('"', '""') + '"'
  
  
  # ============================================================================
  # 直接 SQL 执行支持
  # ============================================================================
  
  _TABLE_NAME_RE = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*$")
  
  
  def _check_ident(name: str) -> str:
      """表名/列名白名单校验，避免拼接 SQL 时的注入风险。"""
      if not isinstance(name, str) or not _TABLE_NAME_RE.match(name):
          raise ValueError(f"Invalid identifier: {name!r}")
      return name
  
  
  def _adapt_sql(sql: str, dialect: str) -> str:
      """
      把数据库无关的 '?' 占位符适配到目标方言：
        - sqlite:   原样使用 '?'
        - mysql:    '?' -> '%s'
        - postgres: '?' -> '$1', '$2', ...
  
      注意：本函数只做简单替换，字符串字面量里的 '?' 会被误替换。
      生产环境若 SQL 里含字面 '?'，请直接使用对应驱动的原生占位符并调用
      execute_sql(sql, params, adapt=False)。
      """
      if dialect == "sqlite":
          return sql
      if dialect == "mysql":
          return sql.replace("?", "%s")
      if dialect == "postgres":
          counter = [0]
  
          def _repl(_m):
              counter[0] += 1
              return f"${counter[0]}"
  
          return re.sub(r"\?", _repl, sql)
      return sql
  
  
  async def execute_sql(sql: str, params: Optional[list] = None,
                        adapt: bool = True) -> int:
      """
      执行一条写操作 SQL（INSERT / UPDATE / DELETE / DDL）。
  
      参数:
        - sql:    SQL 语句，占位符用 '?'（adapt=True 时自动适配方言）
        - params: 占位符对应的参数列表
        - adapt:  是否自动适配占位符。False 时按当前驱动的原生占位符处理
  
      返回:
        受影响的行数。驱动不提供时返回 -1。
      """
      conn = get_database_connection()
      dialect = get_database_category()
      if adapt:
          sql = _adapt_sql(sql, dialect)
      try:
          return await conn.execute_query(sql, params or [])
      except Exception as e:
          raise RuntimeError(f"execute_sql failed: {e}\nSQL: {sql}\nPARAMS: {params}") from e
  
  
  async def select_sql(sql: str, params: Optional[list] = None,
                       adapt: bool = True) -> list:
      """
      执行 SELECT，返回 list[dict]。每行是 {列名: 值} 的字典。
      """
      conn = get_database_connection()
      dialect = get_database_category()
      if adapt:
          sql = _adapt_sql(sql, dialect)
      try:
          return await conn.execute_query_dict(sql, params or [])
      except Exception as e:
          raise RuntimeError(f"select_sql failed: {e}\nSQL: {sql}\nPARAMS: {params}") from e
  
  
  async def select_one(sql: str, params: Optional[list] = None,
                       adapt: bool = True) -> Optional[dict]:
      """返回第一行 dict，无结果返回 None。"""
      rows = await select_sql(sql, params, adapt=adapt)
      return rows[0] if rows else None
  
  
  async def select_value(sql: str, params: Optional[list] = None,
                         adapt: bool = True):
      """返回第一行第一列的值，无结果返回 None。"""
      row = await select_one(sql, params, adapt=adapt)
      if not row:
          return None
      return next(iter(row.values()))
  
  
  async def execute_many(sql: str, params_list: list,
                         adapt: bool = True) -> int:
      """
      批量执行同一条 SQL，参数列表逐条传入。
      返回成功执行的条数。
      """
      conn = get_database_connection()
      dialect = get_database_category()
      if adapt:
          sql = _adapt_sql(sql, dialect)
      ok = 0
      for params in params_list:
          try:
              await conn.execute_query(sql, params or [])
              ok += 1
          except Exception as e:
              raise RuntimeError(
                  f"execute_many failed at {ok + 1}/{len(params_list)}: {e}\n"
                  f"SQL: {sql}\nPARAMS: {params}"
              ) from e
      return ok
  
  
  async def execute_script(script: str) -> bool:
      """
      执行多语句 SQL 脚本（分号分隔）。
      - SQLite: 走 executescript，语义最完整
      - 其它: 按 ';' 拆分逐条执行
      """
      conn = get_database_connection()
      dialect = get_database_category()
  
      if dialect == "sqlite":
          # Tortoise 内部持有的 aiosqlite 连接
          raw = getattr(conn, "_connection", None)
          if raw is not None and hasattr(raw, "executescript"):
              await raw.executescript(script)
              return True
  
      stmts = [s.strip() for s in script.split(";") if s.strip()]
      for s in stmts:
          await execute_sql(s, adapt=False)
      return True
  
  
  async def execute_sql_file(fpath: str) -> bool:
      """从文件加载并执行 SQL 脚本。"""
      if not os.path.isfile(fpath):
          raise FileNotFoundError(f"SQL file not found: {fpath}")
      with open(fpath, "r", encoding="utf-8") as f:
          script = f.read()
      return await execute_script(script)
  
  
  # ---------------- 事务 ----------------
  @asynccontextmanager
  async def transaction(connection_name: str = "default"):
      """
      事务上下文：
  
          async with transaction() as tx:
              await tx.execute_query("INSERT ...", [...])
              await tx.execute_query("UPDATE ...", [...])
          # 正常退出自动 commit，异常自动 rollback
      """
      async with in_transaction(connection_name) as tx:
          yield tx

  async def list_tables() -> list:
      """返回当前数据库中的所有表名。"""
      dialect = get_database_category()
      if dialect == "sqlite":
          rows = await select_sql(
              "SELECT name FROM sqlite_master "
              "WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"
          )
      elif dialect == "mysql":
          rows = await select_sql(
              "SELECT table_name FROM information_schema.tables "
              "WHERE table_schema = DATABASE() ORDER BY table_name"
          )
      elif dialect == "postgres":
          rows = await select_sql(
              "SELECT table_name FROM information_schema.tables "
              "WHERE table_schema = current_schema() ORDER BY table_name"
          )
      else:
          return []
      return [next(iter(r.values())) for r in rows]
  
  
  async def table_exists(name: str) -> bool:
      """判断表是否存在。"""
      _check_ident(name)
      tables = await list_tables()
      return name in tables
  
  
  async def describe_table(name: str) -> list:
      """返回表的列信息（list[dict]，字段随方言而异）。"""
      _check_ident(name)
      dialect = get_database_category()
      if dialect == "sqlite":
          return await select_sql(f"PRAGMA table_info('{name}')")
      if dialect == "mysql":
          return await select_sql(f"SHOW COLUMNS FROM `{name}`")
      if dialect == "postgres":
          return await select_sql(
              "SELECT column_name, data_type, is_nullable "
              "FROM information_schema.columns "
              "WHERE table_schema = current_schema() AND table_name = ? "
              "ORDER BY ordinal_position",
              [name],
          )
      return []

  async def _drop_mysql_database(host, port, usr, psw, name):
      # 不指定 db，直接连服务器
      conn = await aiomysql.connect(
          host=host, port=port,
          user=usr, password=psw,
          autocommit=True,
      )
      try:
          async with conn.cursor() as cur:
              # 先杀掉目标库上的其他连接（需要 PROCESS 权限，权限不足就跳过）
              try:
                  await cur.execute(
                      "SELECT id FROM information_schema.processlist "
                      "WHERE db = %s AND id <> CONNECTION_ID()",
                      (name,),
                  )
                  rows = await cur.fetchall()
                  for (pid,) in rows:
                      await cur.execute(f"KILL {int(pid)}")
              except Exception:
                  pass
  
              await cur.execute(
                  f"DROP DATABASE IF EXISTS {_quote_mysql_ident(name)}"
              )
      finally:
          conn.close()
  
  
  async def _drop_postgres_database(host, port, usr, psw, name):
      # 连到默认的 postgres 管理库
      conn = await asyncpg.connect(
          host=host, port=port,
          user=usr, password=psw,
          database="postgres",
      )
      try:
          # 先终止目标库上的其他连接
          await conn.fetch(
              "SELECT pg_terminate_backend(pid) "
              "FROM pg_stat_activity "

      """
      用 sqlite3 CLI 的 .backup 命令备份 SQLite 数据库。
  
      - .backup 使用 SQLite 的在线备份 API，自动合并 WAL 中的未合并数据

        - 异常  sqlite3 执行失败

          return False

          outfile = f"{file}.{_timestamp()}"

          check=False,
      )
      if result.returncode != 0:
          raise RuntimeError(
              f"sqlite3 backup failed: {result.stderr.decode(errors='replace')}"
          )
  
      return True
  
  
  def _backup_mysql_database(host, port, usr, psw, name, outfile: str = None) -> bool:
      """
      用 mysqldump 备份 MySQL 数据库。
      需要安装 > sudo apt install mysql-client
      参数:
        - host/port/usr/psw/name: 连接参数
        - outfile: 输出文件，默认 f"{name}.sql.bak"
  
      返回: True 成功 / 失败抛 RuntimeError
      """

          outfile = f"{name}.{_timestamp()}.sql"

      with tempfile.NamedTemporaryFile(
          mode="w", suffix=".cnf", delete=False
      ) as f:
          f.write("[client]\n")
          f.write(f"password={psw}\n")
          cnf_path = f.name

          cmd = [
              "mysqldump",
              f"--defaults-extra-file={cnf_path}",
              "-h", host,
              "-P", str(port),
              "-u", usr,
              "--single-transaction",     # InnoDB 一致性快照，不锁表
              "--routines",               # 存储过程/函数
              "--triggers",               # 触发器
              "--events",                 # 事件调度
              "--set-gtid-purged=OFF",    # 避免 GTID 干扰恢复
              name,
          ]
          result = subprocess.run(
              cmd,
              capture_output=True,
              check=False,
          )
          if result.returncode != 0:
              raise RuntimeError(
                  f"mysqldump failed: {result.stderr.decode(errors='replace')}"
              )
          with open(outfile, "wb") as f:
              f.write(result.stdout)
      finally:
          try:
              os.unlink(cnf_path)
          except OSError:
              pass
  
      return True

          raise RuntimeError(
              f"pg_dump failed: {result.stderr.decode(errors='replace')}"
          )

  class KeyedModel(models.ModelMeta):
      def __new__(cls, name, bases, attrs):
          if "id" not in attrs:
              attrs['id'] = fields.CharField(max_length=255, pk=True)
          if '__str__' not in attrs:
              def auto_str(self):
                  return f"{name}({self.id})"
              attrs['__str__'] = auto_str
          return super().__new__(cls, name, bases, attrs)
  
  
  # 层次模型
  class TieredModel(KeyedModel):
      def __new__(cls, name, bases, attrs):
  
          meta_class = attrs.get("Meta")
          fk_mapping = getattr(meta_class, "fk_mapping", {}) if meta_class else {}
  
          # acquire app lablel
          app_label = getattr(meta_class, "app", "models") if meta_class else "models"
  
          # reference parent
          if "parent" not in attrs and "parent" not in fk_mapping:
              attrs['parent'] = fields.ForeignKeyField(
                  f"{app_label}.{name}",  # like "models.ConfigModel"
                  null=True,
                  on_delete=fields.SET_NULL,
                  related_name="+",               # No reverse access
              )
  
          # handle fields in fk_mapping
          for field_name, related_model in fk_mapping.items():
              if field_name not in attrs:
                  attrs[field_name] = fields.ForeignKeyField(
                      related_model,
                      null=True,
                      on_delete=fields.SET_NULL,
                  )
  
          return super().__new__(cls, name, bases, attrs)

  async def get_service_status_text_async(service_name, lines: int = 20,
                             timeout: int = 5) -> str:
      """
      用 tornado.process.Subprocess 异步执行 `systemctl status <id>`，
      完全基于 Tornado 事件循环，不占用任何线程。
  
      为避免 systemctl 在某些环境下挂住不退出，命令前置 `timeout` 工具，
      强杀超时；并在 finally 中显式回收子进程，避免残留。
      """
      proc = tornado.process.Subprocess(
          ["timeout", str(timeout),
           "systemctl", "status", service_name,
           "-n", str(lines), "--no-pager"],
          stdout=tornado.process.Subprocess.STREAM,
          stderr=tornado.process.Subprocess.STREAM,
      )
      try:
          # 并发读两个流，避免某一端缓冲区满导致死锁
          out_b, err_b = await tornado.gen.multi([
              proc.stdout.read_until_close(),
              proc.stderr.read_until_close(),
          ])
      finally:
          # 确保子进程被回收，避免僵尸/残留
          try:
              await proc.wait_for_exit()
          except Exception:
              pass
      out = (out_b or b"").decode("utf-8", errors="replace")
      err = (err_b or b"").decode("utf-8", errors="replace")
      # systemctl status 在服务未运行时退出码非 0，但 stdout 仍有内容
      return out or err

              return False

  def wait_for_service_ready(srv_url: str, timeout: int = 60,
                             interval: float = 0.5,
                             verify: bool = True) -> bool:
      """
      轮询 /ping 直到服务就绪或超时。
  
      建议传入 127.0.0.1 基地址（见 get_local_url），避免依赖云厂商 NAT
      hairpin（从服务器自身访问公网 IP）。
      HTTPS 场景下传入 verify=False，否则会因证书 SAN 不含 127.0.0.1 失败。
      """
      deadline = time.time() + timeout
      attempt = 0
      last_err = None
      while time.time() < deadline:
          attempt += 1
          try:
              resp = requests.get(srv_url + "/ping", timeout=2, verify=verify)
              if resp.status_code == 200:
                  logger.info(f"Service at {srv_url} is ready "
                              f"(attempt {attempt}, verify={verify})")
                  return True
              last_err = f"HTTP {resp.status_code}"
          except Exception as e:
              last_err = f"{type(e).__name__}: {e}"
          time.sleep(interval)
      logger.error(f"Service at {srv_url} not ready after {timeout}s; "
                   f"attempts={attempt}, last error={last_err}")
      return False
  
  
  def wait_for_service_stopped(service_name: str, timeout: int = 30,
                               interval: float = 0.5) -> bool:
      """轮询 systemctl is-active，直到服务不再 active 或超时"""

          result = subprocess.run(
              ["systemctl", "is-active", service_name],
              capture_output=True, text=True, check=False

          if state not in ("active", "activating", "deactivating"):
              logger.info(f"Service {service_name} is stopped (state={state})")
              return True
          time.sleep(interval)
      logger.error(f"Service {service_name} still active after {timeout}s")
      return False

          gServer = XSpawner.getServer()

          await q.put((future, condition, path, headers, body))
  
          cond_res = await condition.wait(timeout=datetime.timedelta(seconds=60))

                  self.write(fdata)
              else:
                  self.set_header('Content-Type', content_type)
                  self.write(result if isinstance(result, str)
                             else json.dumps(result))
          else:
              self.write('false')
          self.finish()

  async def caddy_find_server_by_site_address(site_address: str) -> Optional[str]:
      """
      根据 Caddyfile 中的站点地址（如 "https://example.com"）查找对应的 server_name。
  
      :param site_address: 站点地址，可带协议（https://example.com）或纯域名（example.com）
      :return: server_name（如 "srv0"）；未找到时返回 None
      """
      # 1. 解析出域名（host）
      domain = _extract_domain(site_address)
      if not domain:
          return None
  
      # 2. 拉取全部 server 配置
      try:
          resp = await _fetch("GET", f"{CADDY_ADMIN}/config/apps/http/servers")
          if resp.code >= 400:
              return None
          servers = json.loads(resp.body or b"{}") or {}
      except Exception:
          traceback.print_exc()
          return None
  
      # 3. 遍历每个 server，检查其 routes 中的 host 匹配器
      for server_name, cfg in servers.items():
          try:
              routes = cfg.get("routes") or []
              if _domain_in_routes(routes, domain):
                  return server_name
          except Exception:
              traceback.print_exc()
              continue
  
      return None
  
  
  def _extract_domain(site_address: str) -> Optional[str]:
      """
      从站点地址中提取域名。
      支持 "https://example.com"、"http://example.com"、"example.com"、":443" 等形式。
      """
      site_address = site_address.strip()
      if not site_address:
          return None
  
      # 如果看起来像 URL（含 ://），用 urlparse
      if "://" in site_address:
          parsed = urlparse(site_address)
          return parsed.hostname.lower() if parsed.hostname else None
  
      # 纯域名或带端口的域名
      # 去掉可能存在的端口（如 "example.com:8080"）
      if ":" in site_address:
          # 纯端口（如 ":443"）表示无域名，返回 None
          host_part = site_address.split(":", 1)[0]
          if not host_part:
              return None
          return host_part.lower()
  
      return site_address.lower()
  
  
  def _domain_in_routes(routes: list, domain: str) -> bool:
      for route in routes or []:
          for matcher in route.get("match") or []:
              for host in matcher.get("host") or []:
                  if _host_matches(host, domain):
                      return True
          for handler in route.get("handle") or []:
              if handler.get("handler") == "subroute":
                  if _domain_in_routes(handler.get("routes") or [], domain):
                      return True
              elif handler.get("routes"):
                  if _domain_in_routes(handler.get("routes") or [], domain):
                      return True
      return False
  
  
  def _host_matches(pattern: str, domain: str) -> bool:
      pattern, domain = pattern.strip().lower(), domain.strip().lower()
      if pattern == domain:
          return True
      if not pattern or "*" not in pattern:
          return False
      if pattern.startswith("*."):
          suffix = pattern[2:]
          return domain.endswith("." + suffix) and domain.count(".") == suffix.count(".") + 1
      if pattern.endswith("*"):
          return domain.startswith(pattern[:-1])
      return False

  async def _get_routes(server_name):
      resp = await _fetch("GET", _routes_url(server_name))
      if resp.code >= 400:
          raise RuntimeError(f"GET routes failed: {resp.code}")
      data = json.loads(resp.body or b"null")

  # ── 查询：遍历所有反向代理 ──────────────────────────────────
  async def _walk_routes(routes, inherit_paths=None, inherit_strip=None):
      """
      递归展开 routes，抽取所有 reverse_proxy 条目。
      - subroute 内的子路由若自身没有 match，则继承父路径
      - rewrite / uri 的剥离前缀沿继承链向下传递
      返回 list[dict]: {id, paths, strip, upstream}
      """
      inherit_paths = inherit_paths or []
      out = []
  
      for route in routes or []:
          own_paths = []
          for m in route.get("match") or []:
              own_paths.extend(m.get("path") or [])
          paths = own_paths or inherit_paths
  
          strip = inherit_strip
          upstreams = []
          sub_routes = None
  
          for h in route.get("handle") or []:
              name = h.get("handler")
              if name == "rewrite":
                  strip = h.get("strip_path_prefix") or strip
              elif name == "uri":
                  strip = h.get("strip_prefix") or strip
              elif name == "reverse_proxy":
                  for u in h.get("upstreams") or []:
                      if u.get("dial"):
                          upstreams.append(u["dial"])
              elif name == "subroute":
                  sub_routes = h.get("routes") or []
  
          if sub_routes:
              out.extend(await _walk_routes(sub_routes, paths, strip))
  
          for up in upstreams:
              out.append({
                  "id":       route.get("@id"),
                  "paths":    list(paths),
                  "strip":    strip,
                  "upstream": up,
              })
  
      return out

      列出 Caddy 中指定 server 的全部反向代理路由。
  
      :param server_name: 要查询的 server 名（如 srv0）

          server   所属 server 名（如 srv0）
          listen   监听地址列表（如 [":443"]）
          id       路由 @id（无 @id 则为 None）
          paths    匹配的路径列表
          strip    剥离前缀（无则 None）
          upstream 上游 dial 地址（如 127.0.0.1:8081）

          resp = await _fetch(
              "GET", f"{CADDY_ADMIN}/config/apps/http/servers/{server_name}"

              return result
          cfg = json.loads(resp.body or b"{}") or {}
      except Exception:

          return result

          return result

          ",".join(it.get("listen") or []) or "-",

      table = [list(_PROXY_COLUMNS)]
      for it in items:

      return table
  
  
  def format_proxies_text(items: list) -> str:
      """返回纯文本对齐表格，适合日志或命令行输出。"""
      if not items:
          return "(没有任何 reverse_proxy 路由)"
  
      rows = [_proxy_row(it) for it in items]
      widths = [
          max(len(r[i]) for r in (rows + [list(_PROXY_COLUMNS)]))
          for i in range(len(_PROXY_COLUMNS))
      ]
  
      def _fmt(row):
          return "  ".join(str(row[i]).ljust(widths[i]) for i in range(len(row)))
  
      lines = [_fmt(list(_PROXY_COLUMNS)),
               "-" * (sum(widths) + 2 * (len(widths) - 1))]
      lines.extend(_fmt(r) for r in rows)
      return "\n".join(lines)

  async def caddy_add_route(server_name: str, srv_id: str, port: int) -> bool:
      """幂等：先按 @id 删，再插到 routes 头部（优先于 file_server）。"""
      async with _lock:
          try:
              # 1) 幂等清理，404 忽略
              await _fetch("DELETE", f"{CADDY_ADMIN}/id/{_rid(server_name, srv_id)}")
  
              # 2) 读现有 routes 并剔除同名条目
              routes = await _get_routes(server_name)
              routes = [x for x in routes if x.get("@id") != _rid(server_name, srv_id)]
  
              # 3) 插到最前，确保先于 file_server 命中
              routes.insert(0, _route_obj(server_name, srv_id, port))
  
              # 4) 整体 PATCH
              resp = await _fetch(
                  "PATCH", _routes_url(server_name),
                  body=json.dumps(routes).encode("utf-8"),
              )
              if not (200 <= resp.code < 300):
                  print(f"[caddy] PATCH {resp.code}: "
                        f"{(resp.body or b'').decode('utf-8', 'replace')[:1500]}", flush=True)
              return 200 <= resp.code < 300
          except Exception:
              traceback.print_exc()
              return False
  
  
  async def caddy_del_route(server_name: str, srv_id: str) -> bool:
      """按 @id 精确删除，不做索引运算，天然并发安全。"""
      async with _lock:
          try:
              resp = await _fetch("DELETE", f"{CADDY_ADMIN}/id/{_rid(server_name, srv_id)}")
              return resp.code in (200, 404)   # 404 = 本来就没有，视为成功
          except Exception:
              traceback.print_exc()
              return False
  
  
  async def caddy_bootstrap(server_name: str, children: dict) -> bool:
      """
      Caddy 重启 / caddy reload 后动态路由会丢失，
      启动时用当前子服务列表整体重放一次。
      children: {srv_id: port}
      """
      async with _lock:
          try:
              routes = await _get_routes(server_name)
              routes = [x for x in routes
                        if not str(x.get("@id", "")).startswith("xspawner-")]
              for sid, port in children.items():
                  routes.insert(0, _route_obj(server_name, sid, port))
  
              resp = await _fetch(
                  "PATCH", _routes_url(server_name),
                  body=json.dumps(routes).encode("utf-8"),
              )
              return 200 <= resp.code < 300
          except Exception:
              traceback.print_exc()
              return False

  def get_first_level_json(data, pretty=True):
      """
      返回字典第一级属性的 JSON 字符串。
  
      对特殊类型做降级处理：
        - datetime / date / time  → ISO 字符串
        - Decimal                 → float
        - list / tuple / dict     → 递归简化
        - 其他（函数、模块、对象）→ str(value)，失败则退化为 <类型名>
  
      参数:
          data:   字典数据
          pretty: 是否美化输出（缩进 2 空格）
  
      返回:
          JSON 字符串；若 data 非 dict，返回带 error 字段的 JSON
      """
      if not isinstance(data, dict):
          return json.dumps(
              {"error": f"input must be dict, got {type(data).__name__}"},
              indent=2 if pretty else None,
              ensure_ascii=False,
          )
  
      result = {str(k): _simplify(v) for k, v in data.items()}
      return json.dumps(
          result,
          indent=2 if pretty else None,
          ensure_ascii=False,
          default=str,
      )
  
  
  def _simplify(value):
      """把单个值简化为可 JSON 序列化的形式。"""
      # 基本类型（含 None）原样返回
      if value is None or isinstance(value, (bool, int, float, str)):
          return value
  
      # 时间类型 → ISO 字符串
      if isinstance(value, (datetime.datetime, datetime.date, datetime.time)):
          return value.isoformat()
  
      # Decimal → float
      if isinstance(value, Decimal):
          return float(value)
  
      # 容器类型：递归简化；深度异常时退化为类型名
      if isinstance(value, (list, tuple)):
          try:
              return [_simplify(v) for v in value]
          except Exception:
              return f"<{type(value).__name__} len={len(value)}>"
  
      if isinstance(value, dict):
          try:
              return {str(k): _simplify(v) for k, v in value.items()}
          except Exception:
              return f"<dict len={len(value)}>"
  
      # 其他类型（函数、模块、自定义对象等）：尝试 str
      try:
          return str(value)
      except Exception:
          return f"<{type(value).__name__}>"

      async def _do_add(self):
          await self._load_spec()
  
          fields = [
              ('__key__', '键', 'str', ''),

          if not self.spec:
              fields.append(('__value__', '值', 'str', ''))

              for i, (sk, sv) in enumerate(self.spec.items()):
                  ftype = self._infer_type(None, sv)
                  initial = False if ftype == 'bool' else None
                  fields.append((f'f{i}', sk, ftype, initial))
              indent_after = 1   # 复合数据：主键之后整体缩进
  
          values = await self._show_form_popup(
              '新增数据', fields, indent_after=indent_after
          )
          if values is None:
              return
  
          key = str(values.get('__key__', '') or '').strip()
          if not key:
              toast('键不能为空', color='error'); return
          if key == SPEC_KEY:
              toast(f'"{SPEC_KEY}" 是保留键，请换一个', color='error'); return
  
          if not self.spec:
              value = self._coerce(values.get('__value__'), 'str')
          else:
              value = {}
              for i, (sk, sv) in enumerate(self.spec.items()):
                  ftype = self._infer_type(None, sv)
                  raw = values.get(f'f{i}')
                  value[sk] = self._coerce(raw, ftype)

      # ---------------- 编辑（弹窗） ----------------
      async def _do_edit(self, key):
          if key == SPEC_KEY:
              toast(f'"{SPEC_KEY}" 为保留键，请用 📋 视图编辑', color='error')
              return
  
          old = await self.getValue(key)
          if old is None:
              toast(f'键 "{key}" 不存在', color='error'); return
  
          await self._load_spec()
  
          fields = []
          is_dict = isinstance(old, dict)
  
          if is_dict:
              names = list(self.spec.keys()) if self.spec else []
              for k in old.keys():
                  if k not in names:
                      names.append(k)
              for i, sk in enumerate(names):
                  sv = old.get(sk, self.spec.get(sk))
                  spec_v = self.spec.get(sk)
                  ftype = self._infer_type(sv, spec_v)
                  fields.append((f'f{i}', sk, ftype, sv))
          else:
              ftype = self._infer_type(old, None)
              fields.append(('f0', '值', ftype, old))
  
          # 编辑时主键已作为标题显示，表单里只有内部字段，不缩进
          values = await self._show_form_popup(f'编辑 "{key}"', fields)
          if values is None:
              return
  
          if is_dict:
              value = {}
              for safe, label, ftype, _v in fields:
                  raw = values.get(safe)
                  value[label] = self._coerce(raw, ftype)
          else:
              safe, _label, ftype, _v = fields[0]
              value = self._coerce(values.get(safe), ftype)
  
          ok = await self.setValue(key, value)
          toast('更新成功' if ok else '更新失败',
                color='success' if ok else 'error')
          await self._render_home()
